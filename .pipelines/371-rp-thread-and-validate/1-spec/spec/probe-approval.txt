# 1. First review turn: lanes a and b each open a review thread with a finding

$ node rp.mjs next <repo>/.pipelines/demo --apply
frontier review a
frontier review b
hook before-creating-worktree reviewer-a <repo>/.worktrees/demo-review-a
hook before-creating-worktree reviewer-b <repo>/.worktrees/demo-review-b
create-branch demo-review-a change
create-branch demo-review-b change
create-worktree reviewer-a demo-review-a <repo>/.worktrees/demo-review-a
create-worktree reviewer-b demo-review-b <repo>/.worktrees/demo-review-b
hook after-creating-worktree reviewer-a <repo>/.worktrees/demo-review-a
hook after-creating-worktree reviewer-b <repo>/.worktrees/demo-review-b
hook phase-started 4
hook before-spawning-agent reviewer-a
wake reviewer-a profile reviewer
wake reviewer-a lane a
wake reviewer-a phase 4
wake reviewer-a worktree <repo>/.worktrees/demo-review-a
wake reviewer-a assignment <repo>/.git/worktrees/demo-review-a/rp/reviewer.md
hook after-spawning-agent reviewer-a
hook before-merging-lanes reviewer-a
hook after-merging-lanes reviewer-a
hook before-spawning-agent reviewer-b
wake reviewer-b profile reviewer
wake reviewer-b lane b
wake reviewer-b phase 4
wake reviewer-b worktree <repo>/.worktrees/demo-review-b
wake reviewer-b assignment <repo>/.git/worktrees/demo-review-b/rp/reviewer.md
hook after-spawning-agent reviewer-b
hook before-merging-lanes reviewer-b
hook after-merging-lanes reviewer-b
[exit 0]

$ node rp.mjs land <repo>/.pipelines/demo --role reviewer-a

[exit 0]

$ node rp.mjs land <repo>/.pipelines/demo --role reviewer-b

[exit 0]
